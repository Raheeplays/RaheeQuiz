STEP 1: Add required permissions to your "app/src/main/AndroidManifest.xml":
------------------------------------------------------------------------------
Add these lines inside the `<manifest>` tag near other permission definitions:

    <uses-permission android:name="android.permission.INTERNET" />
    <uses-permission android:name="android.permission.WRITE_EXTERNAL_STORAGE" android:maxSdkVersion="28" />
    <uses-permission android:name="android.permission.READ_EXTERNAL_STORAGE" />


STEP 2: Attach the Download Listener inside your "MainActivity.java" onCreate()
------------------------------------------------------------------------------
Locate where your `myWebView` variable is initialized (usually inside onCreate After setting JavaScript Enabled), 
and copy-paste this direct attachment logic:

```java
        // Enable file downloading natively
        myWebView.setDownloadListener(new android.webkit.DownloadListener() {
            @Override
            public void onDownloadStart(String url, String userAgent, String contentDisposition, String mimeType, long contentLength) {
                // Instantly delegate download task to our custom robust helper
                WebViewDownloadHelper.handleDownload(MainActivity.this, url, userAgent, contentDisposition, mimeType);
            }
        });
```


STEP 3: Add the Helper Class to your Source Directory:
------------------------------------------------------------------------------
Create a new file named "WebViewDownloadHelper.java" in your project source directory under your package hierarchy:
Path: app/src/main/java/RaheeQuiz/in/WebViewDownloadHelper.java

Paste this complete, premium, self-contained Java source code:

```java
package RaheeQuiz.in;

import android.app.Activity;
import android.app.DownloadManager;
import android.content.ContentValues;
import android.content.Context;
import android.net.Uri;
import android.os.Build;
import android.os.Environment;
import android.provider.MediaStore;
import android.util.Base64;
import android.webkit.URLUtil;
import android.widget.Toast;
import java.io.File;
import java.io.FileOutputStream;
import java.io.OutputStream;

/**
 * Robust native helper that intercepts clicks on downloadable links inside WebViews.
 * 
 * Fits perfectly on both older APIs and modern Android 10+ (Scoped Storage/MediaStore).
 * Fully supports base64-encoded client-side PDFs (e.g. data:application/pdf;base64,...)
 * generated by frameworks like jsPDF without crashing the App.
 */
public class WebViewDownloadHelper {

    public static void handleDownload(final Activity activity, final String url, final String userAgent, 
                                      final String contentDisposition, final String mimeType) {
        
        if (url == null || url.trim().isEmpty()) {
            return;
        }

        // --- CASE 1: Client-Side Base64 Data URI (jspdf, certificates, OMR sheet) ---
        if (url.startsWith("data:")) {
            activity.runOnUiThread(new Runnable() {
                @Override
                public void run() {
                    downloadBase64DataUri(activity, url, mimeType);
                }
            });
            return;
        }

        // --- CASE 2: Standard Server-Side HTTP / HTTPS Direct URLs ---
        activity.runOnUiThread(new Runnable() {
            @Override
            public void run() {
                downloadStandardHttpUri(activity, url, userAgent, contentDisposition, mimeType);
            }
        });
    }

    /**
     * Decodes and writes base64 PDF/image data schemes straight to the Downloads folder.
     */
    private static void downloadBase64DataUri(Context context, String base64Url, String mimeType) {
        try {
            // Pick default values if missing or general stream
            if (mimeType == null || mimeType.trim().isEmpty()) {
                mimeType = "application/pdf"; 
            }
            
            String extension = ".pdf";
            if (mimeType.contains("png")) extension = ".png";
            else if (mimeType.contains("jpg") || mimeType.contains("jpeg")) extension = ".jpg";
            else if (mimeType.contains("csv")) extension = ".csv";

            String fileName = "Rahee_Download_" + System.currentTimeMillis() + extension;

            // Strip prefix from schema (e.g. "data:application/pdf;base64,iVBORw0K...")
            int commaIndex = base64Url.indexOf(",");
            if (commaIndex == -1) {
                Toast.makeText(context, "Invalid data package URL", Toast.LENGTH_SHORT).show();
                return;
            }
            String base64Content = base64Url.substring(commaIndex + 1);
            byte[] fileBytes = Base64.decode(base64Content, Base64.DEFAULT);

            // Use scoped storage MediaStore write block for Android 10+ (API 29+) to bypass runtime write permissions
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                ContentValues values = new ContentValues();
                values.put(MediaStore.MediaColumns.DISPLAY_NAME, fileName);
                values.put(MediaStore.MediaColumns.MIME_TYPE, mimeType);
                values.put(MediaStore.MediaColumns.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS);

                Uri collectionUri = MediaStore.Downloads.EXTERNAL_CONTENT_URI;
                Uri fileUri = context.getContentResolver().insert(collectionUri, values);

                if (fileUri != null) {
                    OutputStream os = context.getContentResolver().openOutputStream(fileUri);
                    if (os != null) {
                        os.write(fileBytes);
                        os.flush();
                        os.close();
                        Toast.makeText(context, "Saved/Downloaded: " + fileName, Toast.LENGTH_LONG).show();
                    }
                } else {
                    throw new Exception("Could not register MediaStore download item");
                }
            } else {
                // Fallback writing method for older Android versions (< 10) requiring legacy manifest permissions
                File path = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS);
                if (!path.exists()) {
                    path.mkdirs();
                }
                File file = new File(path, fileName);
                FileOutputStream fos = new FileOutputStream(file);
                fos.write(fileBytes);
                fos.flush();
                fos.close();
                Toast.makeText(context, "Downloaded into directory: " + file.getAbsolutePath(), Toast.LENGTH_LONG).show();
            }
        } catch (Exception e) {
            e.printStackTrace();
            Toast.makeText(context, "Error decoding file: " + e.getMessage(), Toast.LENGTH_LONG).show();
        }
    }

    /**
     * Downloads files securely over standard HTTP/HTTPS channels using the Android DownloadManager class.
     */
    private static void downloadStandardHttpUri(Context context, String url, String userAgent, 
                                                 String contentDisposition, String mimeType) {
        try {
            DownloadManager.Request request = new DownloadManager.Request(Uri.parse(url));
            request.setMimeType(mimeType);

            // Secure web headers
            String cookies = android.webkit.CookieManager.getInstance().getCookie(url);
            request.addRequestHeader("cookie", cookies);
            request.addRequestHeader("User-Agent", userAgent);

            // Build beautiful user status bar interface progress
            request.setDescription("Downloading file from Rahee...");
            String filename = URLUtil.guessFileName(url, contentDisposition, mimeType);
            request.setTitle(filename);

            // Allow scanning by MediaScanner (older API)
            if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q) {
                request.allowScanningByMediaScanner();
            }

            // Visible icon trigger in status screen
            request.setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED);
            
            // Set local destination path
            request.setDestinationInExternalPublicDir(Environment.DIRECTORY_DOWNLOADS, filename);

            // Dispatch download intent directly to System Daemon
            DownloadManager manager = (DownloadManager) context.getSystemService(Context.DOWNLOAD_SERVICE);
            if (manager != null) {
                manager.enqueue(request);
                Toast.makeText(context, "Started Download: " + filename, Toast.LENGTH_LONG).show();
            }
        } catch (Exception e) {
            e.printStackTrace();
            Toast.makeText(context, "Download failed: " + e.getMessage(), Toast.LENGTH_LONG).show();
        }
    }
}
```
